def decrypt_004(payload, key):
    parts = payload.strip().split(":")
    if len(parts) < 4 or parts[0] != "004":
        raise ValueError("not a Standard Notes 004 payload")

    nonce = bytes.fromhex(parts[1])
    ciphertext = base64.b64decode(parts[2])
    associated_data = parts[3].encode()

    return crypto_aead_xchacha20poly1305_ietf_decrypt(
        ciphertext,
        associated_data,
        nonce,
        key,
    )


def main():

import argparse
import csv
import struct
from pathlib import Path

DEFAULT_GGUF = Path(

VALUE_TYPES = {
    0: "UINT8",
    1: "INT8",
    2: "UINT16",
    3: "INT16",
    4: "UINT32",
    5: "INT32",
    6: "FLOAT32",
    7: "BOOL",
    8: "STRING",
    9: "ARRAY",
    10: "UINT64",
    11: "INT64",
    12: "FLOAT64",
}

TOKEN_TYPES = {
    1: "NORMAL",
    2: "UNKNOWN",
    3: "CONTROL",
    4: "USER_DEFINED",
    5: "UNUSED",
    6: "BYTE",
}


def read(fmt, f):
    size = struct.calcsize(fmt)
    data = f.read(size)
    if len(data) != size:
        raise EOFError("unexpected end of file")
    return struct.unpack(fmt, data)[0]


def read_string(f):
    length = read("<Q", f)
    data = f.read(length)
    if len(data) != length:
        raise EOFError("unexpected end of file while reading string")
    return data.decode("utf-8", "replace")


def read_value(f, value_type):
    if value_type == 0:
        return read("<B", f)
    if value_type == 1:
        return read("<b", f)
    if value_type == 2:
        return read("<H", f)
    if value_type == 3:
        return read("<h", f)
    if value_type == 4:
        return read("<I", f)
    if value_type == 5:
        return read("<i", f)
    if value_type == 6:
        return read("<f", f)
    if value_type == 7:
        return bool(read("<?", f))
    if value_type == 8:
        return read_string(f)
    if value_type == 10:
        return read("<Q", f)
    if value_type == 11:
        return read("<q", f)
    if value_type == 12:
        return read("<d", f)
    if value_type == 9:
        element_type = read("<I", f)
        count = read("<Q", f)
        return [read_value(f, element_type) for _ in range(count)]

    raise ValueError(f"unknown GGUF value type: {value_type}")


def parse_gguf_metadata(path):
    with path.open("rb") as f:
        magic = f.read(4)
        if magic != b"GGUF":
            raise ValueError(f"not a GGUF file: magic={magic!r}")

        version = read("<I", f)
        tensor_count = read("<Q", f)
        metadata_count = read("<Q", f)

        metadata = {}
        metadata_types = {}

        for _ in range(metadata_count):
            key = read_string(f)
            value_type = read("<I", f)
            value = read_value(f, value_type)
            metadata[key] = value
            metadata_types[key] = VALUE_TYPES.get(value_type, str(value_type))

    return {
        "version": version,
        "tensor_count": tensor_count,
        "metadata_count": metadata_count,
        "metadata": metadata,
        "metadata_types": metadata_types,
    }


def export_tokens(info, output_path):
    metadata = info["metadata"]
    tokens = metadata.get("tokenizer.ggml.tokens")
    token_types = metadata.get("tokenizer.ggml.token_type")

    if not isinstance(tokens, list):
        raise ValueError("tokenizer.ggml.tokens not found or not an array")

    with output_path.open("w", encoding="utf-8", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(["token_id", "token", "token_type", "token_type_name"])

        for token_id, token in enumerate(tokens):
            token_type = ""
            token_type_name = ""

            if isinstance(token_types, list) and token_id < len(token_types):
                token_type = token_types[token_id]
                token_type_name = TOKEN_TYPES.get(token_type, "")

            writer.writerow([token_id, token, token_type, token_type_name])


def print_summary(info):
    metadata = info["metadata"]
    tokens = metadata.get("tokenizer.ggml.tokens", [])
    token_types = metadata.get("tokenizer.ggml.token_type", [])

    print("GGUF 信息")
    print("version:", info["version"])
    print("tensor_count:", info["tensor_count"])
    print("metadata_count:", info["metadata_count"])
    print("token_count:", len(tokens))

    print("\n关键 tokenizer 元数据")
    for key in [
        "tokenizer.ggml.model",
        "tokenizer.ggml.pre",
        "tokenizer.ggml.bos_token_id",
        "tokenizer.ggml.eos_token_id",
        "tokenizer.ggml.padding_token_id",
        "tokenizer.ggml.seperator_token_id",
    ]:
        value = metadata.get(key)
        if isinstance(value, int) and isinstance(tokens, list) and 0 <= value < len(tokens):
            print(f"{key}: {value} -> {tokens[value]!r}")
        else:
            print(f"{key}: {value!r}")

    print("\n最后 30 个 token")
    start = max(0, len(tokens) - 30)
    for token_id in range(start, len(tokens)):
        token_type = ""
        token_type_name = ""
        if isinstance(token_types, list) and token_id < len(token_types):
            token_type = token_types[token_id]
            token_type_name = TOKEN_TYPES.get(token_type, "")
        print(f"{token_id}\t{tokens[token_id]!r}\t{token_type}\t{token_type_name}")


def main():
    parser = argparse.ArgumentParser(
        description="导出 GGUF 文件中的 tokenizer.ggml.tokens 全量 token id 和 token 值"
    )
    parser.add_argument(
        "gguf",
        nargs="?",
        default=str(DEFAULT_GGUF),
        help="GGUF 模型路径，默认使用当前检材中的 ultraman-663M-BF16.gguf",
    )
    parser.add_argument(
        "-o",
        "--output",
        default="gguf_tokens2.csv",

    output_path = Path(args.output)
